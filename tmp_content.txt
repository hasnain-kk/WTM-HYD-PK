    const App = () => {
      const getHashRoute = () => {
        const hash = window.location.hash.replace('#', '');
        return hash || '/';
      };

      const [route, setRouteState] = useState(getHashRoute());
      
      const loadData = (key, defaults) => {
        const saved = localStorage.getItem(key);
        return saved ? JSON.parse(saved) : defaults;
      };

      const [teamData, setTeamData] = useState(loadData('wtm_team', sharedTeamData));
      const [eventData, setEventData] = useState(loadData('wtm_events', sharedEventData));
      const [blogPosts, setBlogPosts] = useState(loadData('wtm_blogs', [
        { id: 1, type: 'Release', title: 'WTM Hyderabad Announces IWD 2026 Theme', author: 'Hasnain KK', snippet: 'International Women\'s Day returns to Hyderabad as the flagship technical event of the year, focusing on new architectural paradigms and diversity in engineering.', hidden: false },
        { id: 2, type: 'News', title: 'Concentrating Developer Efforts in Sindh', author: 'WTM Core', snippet: 'Mirroring global benchmarks, we are officially expanding our local networking initiatives to create stronger pipelines for junior developers.', hidden: false },
      ]));

      useEffect(() => {
        localStorage.setItem('wtm_team', JSON.stringify(teamData));
      }, [teamData]);
      
      useEffect(() => {
        localStorage.setItem('wtm_events', JSON.stringify(eventData));
      }, [eventData]);

      useEffect(() => {
        localStorage.setItem('wtm_blogs', JSON.stringify(blogPosts));
      }, [blogPosts]);

      useEffect(() => {
        const handleHashChange = () => {
          setRouteState(getHashRoute());
        };
        window.addEventListener('hashchange', handleHashChange);
        return () => window.removeEventListener('hashchange', handleHashChange);
      }, []);

      const setRoute = (newRoute) => {
        if (newRoute === route) return;
        window.location.hash = newRoute;
        setRouteState(newRoute);
        window.scrollTo(0, 0);
      };
      
      const renderRoute = () => {
        if (route === '/') return <HomePage setRoute={setRoute} teamData={teamData} />;
        if (route === '/about') return <AboutPage />;
        if (route === '/team') return <TeamPage setRoute={setRoute} teamData={teamData} />;
        if (route === '/events') return <EventsPage setRoute={setRoute} eventData={eventData} />;
        if (route === '/press') return <PressPage posts={blogPosts} />;
        if (route === '/register') return <RegistrationFormPage />;
        if (route === '/code-of-conduct') return <CodeOfConductPage />;
        if (route === '/admin') return <AdminPage teamData={teamData} setTeamData={setTeamData} eventData={eventData} setEventData={setEventData} blogPosts={blogPosts} setBlogPosts={setBlogPosts} />;
        
        if (route.startsWith('/team/')) {
          const slug = route.split('/')[2];
          return <TeamProfilePage slug={slug} setRoute={setRoute} teamData={teamData} />;
        }
        if (route.startsWith('/events/')) {
          const slug = route.split('/')[2];
          return <EventProfilePage slug={slug} setRoute={setRoute} eventData={eventData} />;
        }
        return <HomePage setRoute={setRoute} teamData={teamData} />;
      };

      return (
        <div>
          <Header setRoute={setRoute} currentRoute={route} />
          <main>
            {renderRoute()}
          </main>
          <Footer setRoute={setRoute} currentRoute={route} />
        </div>
      );
    };
